import numpy as np
class Solution:
    def numSpecial(self, mat: List[List[int]]) -> int:
        arr = np.array(mat)
        sum_rows = np.sum(arr, axis=1)
        sum_cols = np.sum(arr, axis=0)
        check_rows = np.where(sum_rows == 1)[0]
        col_check = np.where(arr[check_rows, :] == 1)[1]
        mask = sum_cols[col_check] == 1
        return int(np.sum(mask))

#Logic

* Redefine the array as a numpy array
* Then sum the rows and columns
* Find all rows that sum to 1
* Find all columns that sum to 1 from the rows that sum to 1
* Return the # of columns that do this
