class Solution:

def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:

	def recur(head: Optional[ListNode], n: int) -> int:
		if head == None:
			return -1
		if head.next == None:
			return n-1
		n = recur(head.next,n)
		if(n>=1):
			return n-1
		elif n == 0:
			head.next = head.next.next
		return -1
	if(recur(head,n)==0):
		if(head.next != None):
			head = head.next
	else:
		head = None
	return head

#Logic

* Using recursion I can backtrack without using extra memory, simply utilizing the stack is good enough
* Lots of edge case checks since my solution covers all ambiguous until lists that are too small such as size 0, 1 ,2
* Recur all the way to the end of the list, then count the amount of backtracks, and return modified list
