import numpy as np
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        h = {}
        for i in range(len(nums)):
            if h.get(target-nums[i]) != None:
                return [h.get(target-nums[i]),i]
            h[nums[i]] = i
        return -1

#Logic

* Use a dict(hashmap) to remember the indices where numbers are: formatted as dict[number] = index
* Then just check if target-nums[i] exists in h, because if it does, the number found is the matching index
