double findMedianSortedArrays(int* nums1, int nums1Size, int* nums2, int nums2Size) {
    int arraypos = 0;
    int numspos = 0;
    int* array = calloc(nums1Size+nums2Size, sizeof(int));
    while(numspos<nums1Size || numspos<nums2Size){
        if(numspos<nums1Size){
            array[arraypos] = nums1[numspos];
            arraypos++;
        }
        if(numspos<nums2Size){
            array[arraypos] = nums2[numspos];
            arraypos++;
        }
        numspos++;
    }
    for(int i = 0;i<nums1Size+nums2Size;i++){
        for(int j = i;j<nums1Size+nums2Size;j++){
            if(array[j]<array[i]){
                int temp = array[j];
                array[j] = array[i];
                array[i] = temp;
            }
        }
    }


    if((nums1Size+nums2Size)%2==0){
        double sum = array[(nums1Size+nums2Size)/2]+array[((nums1Size+nums2Size)/2)-1];
        return sum/2;
    }
    
    return array[(nums1Size+nums2Size)/2];
}
