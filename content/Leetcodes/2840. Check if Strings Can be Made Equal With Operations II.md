import numpy as np
class Solution:
    def checkStrings(self, s1: str, s2: str) -> bool:
        even = np.zeros(26,dtype=int)
        odd = np.zeros(26,dtype=int)
        for index, char in enumerate(s1):
            if index%2 == 0:
                even[ord(char)-ord('a')] += 1
            else:
                odd[ord(char)-ord('a')] +=1
        for index, char in enumerate(s2):
            if index%2 == 0:
                even[ord(char)-ord('a')] -= 1
            else:
                odd[ord(char)-ord('a')] -=1
        return bool(np.all(even == 0) and np.all(odd == 0))

#Logic

* Very naive logic: since all even characters can permeate, and so can all odd characters, count all even and odd characters and check if they are the same


#Complexity 
The complexity is $O(n)$, since we just loop through both of the strings.
