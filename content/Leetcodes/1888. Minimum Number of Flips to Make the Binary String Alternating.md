class Solution:
    def minFlips(self, s: str) -> int:
        n = len(s)
        a = int((1<<n+1)//3)
        x = int(s,2)^a
        m = x.bit_count()
        if n%2 == 0:
            return min(m,n-m)
        diff1 = m
        diff2 = n - m
        for i in range(n):
            if s[i] == '1':
                new_diff1 = diff2 - 1
                new_diff2 = diff1 + 1
            else:
                new_diff1 = diff2 + 1
                new_diff2 = diff1 - 1
            diff1, diff2 = new_diff1, new_diff2
            m = min(m, diff1, diff2)
        return m

#Logic

* This is incredibly similar to a previous problem, but the type-1 operation makes it slightly different
* We can mathematically find the one of the alternating sequences, in our case it starts with a 1
* We use the XOR operation (which finds all differentiating bits)
* On the remaining bit sequence, we just count the # of bits
* To consider the other sequence, we just use n-m, as this is the number of flips needed to find the other sequence
* We can immediately return the minimum if the length of the string is even, as the type-1 operation does nothing in this situation
* For odd strings, we can mathematically calculate the minimum for every type-1 operation, as it simply changes the parity of the displaced numbers we already calculated
* We keep track of the minimum found and just return it once we loop through the entirety of the string
