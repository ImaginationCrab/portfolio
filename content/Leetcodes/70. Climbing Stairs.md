import numpy as np
class Solution:
    def climbStairs(self, n: int) -> int:
        dp = np.zeros(n+1)
        for i in range(n+1):
            if i == 1:
                dp[1] = 1
                continue
            if i == 2:
                dp[2] = 2
                continue
            dp[i] = dp[i-1]+dp[i-2]
        return int(dp[n])
